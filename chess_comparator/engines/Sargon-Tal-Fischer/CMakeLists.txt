cmake_minimum_required(VERSION 3.16)
project(SargonTalFischer LANGUAGES CXX)
if(CMAKE_SIZEOF_VOID_P LESS 8)
  message(FATAL_ERROR "This engine must be built for a 64-bit target")
endif()
find_package(Threads REQUIRED)
add_executable(Sargon-Tal-Fischer-x64 src/main.cpp src/thc.cpp)
target_compile_features(Sargon-Tal-Fischer-x64 PRIVATE cxx_std_17)
target_link_libraries(Sargon-Tal-Fischer-x64 PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(Sargon-Tal-Fischer-x64 PRIVATE /O2 /EHsc /DNDEBUG /D_CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(Sargon-Tal-Fischer-x64 PRIVATE -O3 -DNDEBUG -Wall -Wextra -Wno-stringop-overread)
endif()
