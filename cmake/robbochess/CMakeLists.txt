cmake_minimum_required(VERSION 3.16)
project(RoboChessEngine LANGUAGES C)
set(SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../chess_comparator/robbochess/src")
file(GLOB ENGINE_SOURCES "${SOURCE_DIR}/*.c")
add_executable(RoboChess-x64 ${ENGINE_SOURCES})
target_include_directories(RoboChess-x64 PRIVATE "${SOURCE_DIR}")
target_compile_definitions(RoboChess-x64 PRIVATE _CRT_SECURE_NO_WARNINGS WIN32 _WINDOWS)
target_compile_options(RoboChess-x64 PRIVATE /O2 /W3)
target_link_libraries(RoboChess-x64 PRIVATE winmm)
